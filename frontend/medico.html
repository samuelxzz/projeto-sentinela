async function internarPaciente(index) { const paciente = triagensCarregadas[index]; if (!paciente) { alert("❌ Paciente não encontrado."); return; } const confirmar = confirm( "🏥 Deseja internar o paciente " + paciente.nome + "?" ); if (!confirmar) return; // Abre a aba imediatamente // para o navegador não bloquear a nova aba const novaAba = window.open( "internacao.html", "_blank" ); // Pergunta o número do leito const leito = prompt( "🛏️ Informe o número do leito para " + paciente.nome + ":" ); // Cancelou ou deixou vazio if ( leito === null || String(leito).trim() === "" ) { if ( novaAba && !novaAba.closed ) { novaAba.close(); } alert( "⚠️ Informe o número do leito." ); return; } const numeroLeito = String(leito).trim(); try { const resposta = await fetch( API + "/internacao", { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ paciente: paciente.nome, leito: numeroLeito }) } ); const resultado = await resposta.json(); if (!resposta.ok) { if ( novaAba && !novaAba.closed ) { novaAba.close(); } throw new Error( resultado.erro || "Erro ao realizar a internação." ); } // Atualiza a página da nova aba const url = "internacao.html" + "?paciente=" + encodeURIComponent( paciente.nome ) + "&leito=" + encodeURIComponent( numeroLeito ); if ( novaAba && !novaAba.closed ) { novaAba.location.href = url; } else { window.open( url, "_blank" ); } alert( "✅ Paciente internado com sucesso!\n\n" + "👤 Paciente: " + paciente.nome + "\n" + "🛏️ Leito: " + numeroLeito ); await carregar(); } catch (erro) { console.error( "Erro na internação:", erro ); alert( "❌ Não foi possível realizar a internação.\n\n" + erro.message ); } }
